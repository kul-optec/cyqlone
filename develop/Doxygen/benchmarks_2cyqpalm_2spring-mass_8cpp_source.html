106qp::problems::SpringMassProblem create_problem(const SpringMassParams &params) {
107 auto problem = qp::problems::spring_mass(params);
108 // Add a random disturbance to the initial position of the masses
109 std::seed_seq seed{params.seed, params.seed >> 32};
110 std::mt19937 rng{seed};
111 using dist_t = std::uniform_real_distribution<real_t>;
112 auto dist = params.p_max > 1 ? dist_t{-3, 3} : dist_t{-0.6, -0.1};
113 for (index_t r = 0; r < static_cast<index_t>(params.masses.size()); ++r)
114 problem.ocp.b(0)(r, 0) += dist(rng);
115 return problem;
116}

168void run_benchmark(benchmark::State &state, [[maybe_unused]] const std::string &param_name,
169 const SpringMassParams &params, qp::CyQPALMBackendSettings backend_settings,
170 qp::Settings settings, bool warm = false, bool trace = false) try {
171 disable_tracing(); // No tracing during the actual benchmark
172 if (backend_settings.processors < 1)
173 return state.SkipWithMessage("Number of processors must be at least 1.");
174 auto problem = create_problem(params);
175 // Build a QPALM Cyqlone solver
176 auto ocp = cyqlone::CyqloneStorage<>::build(problem.ocp);
177 auto backend = qp::make_cyqpalm_backend<VL, Order>(ocp, {}, backend_settings);
178 qp::Solver<qp::CyQPALMBackend<VL, Order> *> qpalm{backend.get(), settings};
179 // Main benchmark
180 qp::SolverStatus status;
181 double time_solve = 0, time_active_set_change = 0, time_line_search = 0,
182 time_recompute_outer = 0, time_recompute_inner = 0, time_mat_vec = 0;
183 qpalm(); // Warm up
184 if (warm) {
185 const auto nx = problem.ocp.dim.nx, nu = problem.ocp.dim.nu;
186 qpalm.warm_start_solution();
187 auto x = qpalm.get_solution();
188 std::copy_n(x.data() + nu, nx, problem.ocp.b(0).data);
189 ocp.update(problem.ocp);
190 update_cyqpalm_backend(*qpalm.backend, ocp);
191 }
192 for (auto _ : state) {
193 status = qpalm();
194 state.SetIterationTime(seconds(qpalm.stats->timings.total.wall_time).count());
195 time_solve += seconds(qpalm.stats->timings.solve.wall_time).count();
196 time_active_set_change += seconds(qpalm.stats->timings.active_set_change.wall_time).count();
197 time_line_search += seconds(qpalm.stats->timings.line_search.wall_time).count();
198 time_recompute_outer += seconds(qpalm.stats->timings.recompute_outer.wall_time).count();
199 time_recompute_inner += seconds(qpalm.stats->timings.recompute_inner.wall_time).count();
200 time_mat_vec += seconds(qpalm.stats->timings.mat_vec_A.wall_time).count();
201 time_mat_vec += seconds(qpalm.stats->timings.mat_vec_AT.wall_time).count();
202 time_mat_vec += seconds(qpalm.stats->timings.mat_vec_M.wall_time).count();
203 time_mat_vec += seconds(qpalm.stats->timings.mat_vec_MT.wall_time).count();
204 time_mat_vec += seconds(qpalm.stats->timings.mat_vec_Q.wall_time).count();
205 }
206 if (trace)
207 trace_run([&] { qpalm(); }, param_name);
208 auto sol = ocp.reconstruct_solution(problem.ocp, qpalm.get_solution(),
209 qpalm.get_inequality_multipliers(),
210 qpalm.get_equality_multipliers());
211 auto kkt_error = problem.ocp.compute_kkt_error(sol);
212
213 state.counters["status"] = counter(status);
214 state.counters["success"] = counter(status == qp::SolverStatus::Converged);
215 state.counters["iter"] = counter(qpalm.stats->inner_iter);
216 state.counters["outer_iter"] = counter(qpalm.stats->outer_iter);
217 state.counters["num_factor"] = counter(qpalm.stats_backend->num_factor);
218 state.counters["num_upd"] = counter(qpalm.stats_backend->num_updates);
219 state.counters["rank_upd"] = counter(qpalm.stats_backend->rank_updates);
220 state.counters["t_solve"] = counter_avg(time_solve);
221 state.counters["t_active_set_change"] = counter_avg(time_active_set_change);
222 state.counters["t_line_search"] = counter_avg(time_line_search);
223 state.counters["t_recompute_outer"] = counter_avg(time_recompute_outer);
224 state.counters["t_recompute_inner"] = counter_avg(time_recompute_inner);
225 state.counters["t_mat_vec"] = counter_avg(time_mat_vec);
226 state.counters["res_dual"] = counter(kkt_error.stationarity);
227 state.counters["res_eq"] = counter(kkt_error.equality_residual);
228 state.counters["res_ineq"] = counter(kkt_error.inequality_residual);
229 state.counters["compl"] = counter(kkt_error.complementarity);
230} catch (const std::exception &e) {
231 state.SkipWithError(e.what());
232}

286std::generator<Problem> get_spring_mass_params(const Options &opts) {
287 using std::format;
288 for (auto M : opts.masses)
289 for (auto N : opts.horizon)
290 for (uint64_t seed = opts.seed; seed < opts.seed + opts.num_instances; ++seed)
291 switch (opts.problem_type) {
292 case ProblemType::WangBoyd2008:
293 co_yield {
294 .name = format("spring-mass(M={},N={},seed={}) wb2008", M, N, seed),
295 .params = SpringMassParams::wang_boyd_2008(M, N, seed),
296 };
297 break;
298 case ProblemType::WangBoyd2008Width:
299 co_yield {
300 .name = format("spring-mass(M={},N={},seed={}) wb2008w", M, N, seed),
301 .params = SpringMassParams::wang_boyd_2008_width(M, N, seed, 1),
302 };
303 break;
304 case ProblemType::Domahidi2012:
305 co_yield {
306 .name = format("spring-mass(M={},N={},seed={}) d2012", M, N, seed),
307 .params = SpringMassParams::domahidi_2012(M, N, seed),
308 };
309 break;
310 case ProblemType::ActiveStateConstr:
311 co_yield {
312 .name = format("spring-mass(M={},N={},seed={}) act", M, N, seed),
313 .params = SpringMassParams::active_state_constr(M, N, seed),
314 };
315 break;
316 default: throw std::runtime_error("Unknown problem type");
317 }
318}

342template <index_t VL, qp::StorageOrder O>
343std::generator<Solver> get_cyqlone_solvers(const Options &opts) {
344 const auto cyqlone_solver = [trace{opts.trace}](std::string_view name,
345 const qp::CyQPALMBackendSettings &backend,
346 const qp::Settings &settings) {
347 const auto param_name =
348 std::format("cyqlone(p={},v={},{},{})", backend.processors, VL, order(O), name);
349 return Solver{param_name, [=](benchmark::State &state, const SpringMassParams &params) {
350 run_benchmark<VL, O>(state, param_name, params, backend, settings, true,
351 trace);
352 }};
353 };
354 for (auto p : opts.parallelism) {
355 qp::CyQPALMBackendSettings backend{
356 .processors = p,
357 .changing_constr_factor = opts.changing_constr_factor,
358 .max_update_count = 20,
359 .tricyqle_params =
360 {
361 .pcr_max_update_fraction = opts.pcr_max_update_fraction,
362 .cr_max_update_fraction_Y0 = opts.cr_max_update_fraction,
363 .parallel_solve_cr_threshold = opts.parallel_solve_cr_threshold,
364 .parallel_factor_pcr_threshold = opts.parallel_factor_pcr_threshold,
365 },
366 };
367 qp::Settings settings{
368 .tolerance = 1e-8,
369 .dual_tolerance = 1e-8,
370 .initial_penalty_y = 20,
371 .verbose = false,
372 };
373 qp::CyQPALMBackendSettings backend_no_upd = backend;
374 backend_no_upd.max_update_count = 0;
375 qp::Settings settings_warm = settings;
376 settings_warm.initial_penalty_y = 1e4;
377 settings_warm.initial_inner_tolerance = 1e-4;
378 if (opts.pcr)
379 backend.tricyqle_params.solve_method = cyqlone::SolveMethod::PCR;
380 if (opts.cold) {
381 backend.strategy = backend_no_upd.strategy = qp::WarmStartingStrategy::Zeros;
382 co_yield cyqlone_solver("zero", backend, settings);
383 if (opts.no_updates)
384 co_yield cyqlone_solver("zero,upd=0", backend_no_upd, settings);
385 }
386 if (opts.warm_shift) {
387 backend.strategy = backend_no_upd.strategy = qp::WarmStartingStrategy::Shift;
388 co_yield cyqlone_solver("shift", backend, settings_warm);
389 if (opts.no_updates)
390 co_yield cyqlone_solver("shift,upd=0", backend_no_upd, settings_warm);
391 }
392 if (opts.warm_copy) {
393 backend.strategy = backend_no_upd.strategy = qp::WarmStartingStrategy::Copy;
394 co_yield cyqlone_solver("copy", backend, settings_warm);
395 if (opts.no_updates)
396 co_yield cyqlone_solver("copy,upd=0", backend_no_upd, settings_warm);
397 }
398 }
399}

424template <qp::StorageOrder Order>
425std::generator<Solver> get_cyqlone_solvers_vl(const Options &opts) {
426 static constexpr auto VLs = batmat::types::vl_for_real_t;
427 static constexpr auto to_string = [](auto vl) { return std::to_string(vl); };
428 static constexpr auto lut = guanaqo::make_lut<VLs>([]<index_t VL>(batmat::index_constant<VL>) {
429 return std::make_pair(VL, get_cyqlone_solvers<VL, Order>);
430 });
431 for (auto v : opts.vector_length) {
432 auto dispatch = std::ranges::find(lut, v, &decltype(lut)::value_type::first);
433 if (dispatch == lut.end())
434 throw std::invalid_argument("Unsupported vector length. Supported lengths: " +
435 guanaqo::join(std::views::transform(VLs, to_string)));
436 co_yield elements_of(dispatch->second(opts));
437 }
438}

448auto register_benchmarks(const Options &opts) {
449 size_t max_problem_name_len = 0;
450 size_t max_solver_name_len = 0;
451 for (auto problem : get_spring_mass_params(opts)) {
452 max_problem_name_len = std::max(max_problem_name_len, problem.name.size());
453 for (auto solver : get_solvers(opts)) {
454 max_solver_name_len = std::max(max_solver_name_len, solver.name.size());
455 benchmark::RegisterBenchmark(std::format("{}@{}", problem.name, solver.name),
456 [params = problem.params, run = std::move(solver.run)](
457 benchmark::State &state) { run(state, params); })
458 ->MeasureProcessCPUTime()
459 ->UseManualTime()
460 ->Unit(benchmark::kMillisecond)
461 ->ComputeStatistics("max", [](auto &v) { return *std::ranges::max_element(v); })
462 ->ComputeStatistics("min", [](auto &v) { return *std::ranges::min_element(v); })
463 ->GetName();
464 }
465 }
466 return std::make_pair(max_problem_name_len, max_solver_name_len);
467}
